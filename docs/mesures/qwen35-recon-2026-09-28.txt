def rng(url, a, b):

def size(url):
    out=subprocess.run(["curl","-s","-L","-I","--max-time","60",url],capture_output=True,text=True).stdout
    sizes=[int(m) for m in re.findall(r"(?i)content-length:\s*(\d+)",out)]

class R:
    def __init__(s,b): s.b=b; s.p=0
    def take(s,n):
        if s.p+n>len(s.b): raise EOFError
        v=s.b[s.p:s.p+n]; s.p+=n; return v
    def u32(s): return struct.unpack("<I",s.take(4))[0]
    def u64(s): return struct.unpack("<Q",s.take(8))[0]
    def str(s): n=s.u64(); return s.take(n).decode("utf-8","replace")
SC={0:1,1:1,2:2,3:2,4:4,5:4,6:4,7:1,10:8,11:8,12:8}
FMT={0:"<B",1:"<b",2:"<H",3:"<h",4:"<I",5:"<i",6:"<f",7:"<?",10:"<Q",11:"<q",12:"<d"}
def val(r,t):
